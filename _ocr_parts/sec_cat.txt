def suggest_category(merchant) -> Optional[str]:
    if not merchant:
        return None
    low = merchant.lower()
    best = None
    for cat, kws in _CATEGORY_KEYWORDS.items():
        if any(k in low for k in kws):
            best = cat  # later keys override earlier ones (intentional)
    return best
